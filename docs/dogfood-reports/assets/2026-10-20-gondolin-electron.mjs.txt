import assert from 'node:assert/strict'
import { createRequire } from 'node:module'
import { readFile, writeFile, mkdir } from 'node:fs/promises'
import { execFileSync } from 'node:child_process'
import { performance } from 'node:perf_hooks'
const root = new URL('.', import.meta.url).pathname
const { origin, controller, runner } = JSON.parse(await readFile(root + 'running.json', 'utf8'))
process.kill(controller, 0)
const { _electron } = createRequire('/Users/taymoor/Projects/CIVIC/packages/canvas/package.json')('@playwright/test')
const env = { ...process.env, CIVIC_LOCAL_STORE: root + 'electron-profile-' + Date.now(), CIVIC_DESKTOP_ORIGIN: origin }
delete env.ELECTRON_RUN_AS_NODE
await mkdir(root + 'evidence', { recursive: true })
const samples = []
const sample = () => {
  const rows = execFileSync('ps', ['-axo', 'pid,ppid,rss,pcpu,command'], { encoding: 'utf8' }).split('\n').map(line => { const m = line.trim().match(/^(\d+)\s+(\d+)\s+(\d+)\s+([\d.]+)\s+(.+)$/); return m && { pid: +m[1], ppid: +m[2], rssKiB: +m[3], cpu: +m[4], command: m[5] } }).filter(Boolean)
  const owned = new Set([controller]); for (let i = 0; i < 5; i++) for (const row of rows) if (owned.has(row.ppid)) owned.add(row.pid)
  const group = rows.filter(row => owned.has(row.pid))
  samples.push({ time: Date.now(), rssKiB: group.reduce((sum, row) => sum + row.rssKiB, 0), processes: group })
}
sample()
const interval = setInterval(sample, 500)
const start = performance.now()
const app = await _electron.launch({ executablePath: '/Users/taymoor/Projects/orca/node_modules/.pnpm/electron@44.9.0/node_modules/electron/dist/Electron.app/Contents/MacOS/Electron', args: ['/Users/taymoor/Projects/CIVIC/packages/local/desktop.mjs'], env, timeout: 30000 })
const budget = setTimeout(() => app.process().kill(), 150000)
console.log('LAUNCHED', app.process().pid)
app.process().stderr.on('data', data => console.log('ELECTRON', String(data).slice(0, 2000)))
let result, page
let shellMs, liveMs
const refresh = [], reload = []
const errors = [], failures = []
try {
  console.log('CONTENTS', await app.evaluate(({webContents}) => webContents.getAllWebContents().map(w => ({id:w.id,url:w.getURL(),type:w.getType(),loading:w.isLoading()}))))
  page = await app.firstWindow({timeout:15000})
  console.log('INITIAL', page.url())
  const pending = new Set()
  page.on('request', request => pending.add(request.url()))
  page.on('requestfinished', request => pending.delete(request.url()))
  page.on('requestfailed', request => pending.delete(request.url()))
  setTimeout(() => console.log('PENDING', [...pending].slice(-20)), 8000).unref()
  page.on('pageerror', error => { errors.push(error.message); console.log('PAGE_ERROR', error.message) })
  page.on('console', message => { if (message.type() === 'error') console.log('CONSOLE_ERROR', message.text()) })
  page.on('requestfailed', request => console.log('REQUEST_FAILED', request.url(), request.failure()?.errorText))
  page.on('response', response => { if (response.status() >= 500) { failures.push([response.status(), response.url()]); console.log('HTTP_ERROR',response.status(),response.url()) } })
  await page.waitForLoadState('domcontentloaded', {timeout:15000})
  console.log('PAGE_STATE', await page.evaluate(() => ({ url: location.href, ready: document.readyState, scripts: [...document.scripts].map(element => element.src), desktop: Boolean(window.civicDesktop) })))
  await page.getByLabel('Entire application canvas', { exact: true }).waitFor({ timeout: 20000 })
  shellMs = performance.now() - start
  const liveStart = performance.now()
  await page.goto(origin + '/review?desktop=darwin&scenario=app.home&play=1')
  const frame = page.locator('iframe[title^="Live "]').first()
  await frame.waitFor({ timeout: 60000 })
  await page.waitForFunction(() => document.querySelector('iframe[title^="Live "]')?.contentDocument?.querySelector('[data-rehearsal-ready="true"]'), undefined, { timeout: 60000 })
  const homeReady = () => page.frameLocator('iframe[title^="Live "]').first().getByRole('heading', {name:'Today', exact:true}).waitFor({timeout:20000})
  await homeReady()
  liveMs = performance.now() - liveStart
  console.log('LIVE', { shellMs, liveMs, title: await frame.getAttribute('title') })
  assert.match(await frame.evaluate(element => element.contentDocument.body.textContent), /Today|Closet|Hamper|Laundry/i)
  await frame.evaluate(element => { element.contentWindow.gondolinDocument = 'retained' })
  for (const value of []) {
    const begin = performance.now()
    await writeFile(root + 'source/laundry/apps/web/src/gondolin-refresh.css', `:root { --gondolin-refresh-proof: ${value}; }\n`)
    let hotMs = null
    await page.waitForFunction(value => getComputedStyle(document.documentElement).getPropertyValue('--gondolin-refresh-proof').trim() === String(value), value, { timeout: 3000 }).then(() => {hotMs = performance.now() - begin}).catch(() => {})
    assert.equal(await frame.evaluate(element => element.contentWindow.gondolinDocument), 'retained')
    const reloadStart = performance.now()
    await page.reload({waitUntil:'domcontentloaded'})
    await page.waitForFunction(value => getComputedStyle(document.documentElement).getPropertyValue('--gondolin-refresh-proof').trim() === String(value) && document.querySelector('iframe[title^="Live "]')?.contentDocument?.querySelector('[data-rehearsal-ready="true"]'), value, {timeout:20000})
    refresh.push({hotMs, explicitReloadMs:performance.now()-reloadStart, totalMs:performance.now()-begin})
    await frame.evaluate(element => {element.contentWindow.gondolinDocument = 'retained'})
  }
  for (let i = 0; i < 3; i++) {
    const begin = performance.now()
    await page.reload()
    await page.waitForFunction(() => document.querySelector('iframe[title^="Live "]')?.contentDocument?.querySelector('[data-rehearsal-ready="true"]'), undefined, { timeout: 60000 })
    await homeReady()
    reload.push(performance.now() - begin)
  }
  for (const width of [1440, 390]) {
    await app.evaluate(({ BaseWindow }, width) => BaseWindow.getAllWindows()[0].setSize(width, 900), width)
    assert.equal(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth), true)
    await homeReady()
    await page.screenshot({ path: root + `evidence/electron-${width}.png` })
  }
  sample()
  result = { shellMs, liveMs, refreshMs: refresh, reloadMs: reload, rssPeakMiB: Math.max(...samples.map(row => row.rssKiB)) / 1024, samples, errors, failures, runner }
  await writeFile(root + 'electron.json', JSON.stringify(result, null, 2))
  console.log(JSON.stringify({ ...result, samples: `${samples.length} retained in electron.json` }, null, 2))
  assert.deepEqual(errors, [])
  assert.equal(failures.filter(([,url]) => !url.includes('/fixture.png')).length, 0, 'Core runtime and shell requests succeed; snapshot errors remain separately reported')
} catch (error) {
  sample()
  await writeFile(root + 'electron-partial.json', JSON.stringify({shellMs,liveMs,refresh,reload,rssPeakMiB:Math.max(...samples.map(row=>row.rssKiB))/1024,samples,errors,failures,message:error.message},null,2))
  await writeFile(root + 'electron-failure.json', JSON.stringify({ message: error.message, errors, failures, body: await page?.locator('body').innerText().catch(() => '') }, null, 2))
  await page?.screenshot({ path: root + 'evidence/electron-failure.png' }).catch(() => {})
  throw error
} finally {
  clearInterval(interval)
  await Promise.race([app.evaluate(({ app }) => app.exit(0)).catch(() => {}), new Promise(resolve => setTimeout(resolve, 2000))])
  app.process().kill('SIGKILL')
  clearTimeout(budget)
}
