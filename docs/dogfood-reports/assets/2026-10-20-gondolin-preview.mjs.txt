import assert from 'node:assert/strict'
import { VM, VmCheckpoint, ReadonlyProvider, RealFSProvider, createHttpHooks } from '@earendil-works/gondolin'
import { readFile, writeFile } from 'node:fs/promises'
import { performance } from 'node:perf_hooks'
import { execFileSync } from 'node:child_process'
import { existsSync } from 'node:fs'
const root = new URL('.', import.meta.url).pathname
const input = JSON.parse(await readFile(root + 'source.json', 'utf8'))
const metrics = { startedAt: new Date().toISOString(), phases: {} }
let setup = true
const { httpHooks } = createHttpHooks({ allowedHosts: ['registry.npmjs.org', 'dl-cdn.alpinelinux.org'], blockInternalRanges: true, isRequestAllowed: request => setup && ['GET', 'HEAD'].includes(request.method) && ['registry.npmjs.org', 'dl-cdn.alpinelinux.org'].includes(new URL(request.url).hostname) })
let vm, ingress
process.once('SIGTERM', () => { if (vm) void vm.close() })
const start = performance.now()
const options = () => ({ memory: '3G', cpus: 2, rootfs: { mode: 'cow', size: '8G' }, tmpfs: { '/tmp': { size: '256m' } }, vfs: { mounts: { '/source': new ReadonlyProvider(new RealFSProvider(root + 'source')) } }, httpHooks, startTimeoutMs: 60000, sandbox: { qemuPath: '/opt/homebrew/bin/qemu-system-aarch64' }, sessionLabel: 'CIVIC Laundry disposable spike' })
const prepared = existsSync(root + 'ready.qcow2')
const checkpointPath = root + (prepared ? 'ready.qcow2' : 'dependencies.qcow2')
const resumed = existsSync(checkpointPath)
try {
  vm = resumed ? await VmCheckpoint.load(checkpointPath).resume(options()) : await VM.create(options())
  console.log('BOOT', Math.round(performance.now() - start), vm.id)
  const run = async command => {
    const result = await vm.exec(command)
    if (!result.ok) throw Error(result.stdout + result.stderr)
    return result.stdout
  }
  const readonly = await vm.exec('echo forbidden >> /source/laundry/package.json')
  assert.notEqual(readonly.exitCode, 0)
  metrics.phases.bootReadyMs = performance.now() - start
  console.log('READONLY', readonly.stderr, metrics.phases.bootReadyMs)
  if (!resumed) {
  console.log(await run('mkdir -p /work/project; cp -R /source/. /work/project'))
  console.log('INSTALL_BEGIN')
  const installStart = performance.now()
  const install = vm.exec('npm install --global pnpm@9.15.9 --ignore-scripts --no-audit --no-fund && cd /work/project && pnpm install --ignore-scripts --no-frozen-lockfile', { stdout: 'pipe', stderr: 'pipe' })
  for await (const { stream, text } of install.output()) process.stdout.write(`${stream}: ${text}`)
  assert.equal((await install).exitCode, 0, 'Linux dependency installation succeeds')
  setup = false
  metrics.phases.installMs = performance.now() - installStart
  await writeFile(root + 'metrics.json', JSON.stringify(metrics, null, 2))
  console.log('INSTALL_DONE', Math.round(performance.now() - start))
  const checkpoint = await vm.checkpoint(checkpointPath)
  vm = await checkpoint.resume(options())
  }
  if (!prepared) {
    setup = true
    console.log('BROWSER_SETUP', await run('apk add --no-cache chromium'))
    console.log(await run('test -e /work/project/node_modules/.pnpm/node_modules/tailwindcss; ln -s .pnpm/node_modules/tailwindcss /work/project/node_modules/tailwindcss'))
    const checkpoint = await vm.checkpoint(root + 'ready.qcow2')
    vm = await checkpoint.resume(options())
  }
  setup = false
  console.log(await run('cp /source/guest.mjs /work/project/guest.mjs'))
  for (const [index, path] of input.workspaces.entries()) console.log(await run(`mkdir -p /work/modules/${index}; cp -R /work/project/${path}/node_modules/. /work/modules/${index}; mount --bind /source/${path} /work/project/${path}; mount --bind /work/modules/${index} /work/project/${path}/node_modules`))
  console.log(await run('mount --bind /source/laundry /work/project/laundry; mkdir -p /work/project/adapter/node_modules/.pnpm; mount --bind /work/project/node_modules/.pnpm /work/project/adapter/node_modules/.pnpm'))
  assert.notEqual((await vm.exec('echo forbidden >> /work/project/laundry/package.json')).exitCode, 0)
  ingress = await vm.enableIngress({ listenHost: '127.0.0.1', listenPort: 0 })
  vm.setIngressRoutes([{ prefix: '/', port: 8470, stripPrefix: false }])
  await writeFile(root + 'running.json', JSON.stringify({ origin: ingress.url, id: vm.id, controller: process.pid, runner: vm.getHostPid(), started: Date.now(), resumed, metrics }))
  console.log('INGRESS', ingress.url)
  const server = vm.exec('cd /work/project && node guest.mjs', { stdout: 'pipe', stderr: 'pipe' })
  for await (const { stream, text } of server.output()) {
    process.stdout.write(`${stream}: ${text}`)
    if (text.includes('CIVIC_GONDOLIN_READY')) {
      metrics.phases.serverReadyMs = performance.now() - start
      await writeFile(root + 'warm-start.json', JSON.stringify(metrics, null, 2))
    }
  }
  const result = await server
  throw Error('Preview exited: ' + result.exitCode)
} finally {
  if (ingress) await ingress.close()
  if (vm) await vm.close()
  console.log('CLOSED', Math.round(performance.now() - start))
}
