let files=0
const digest=createHash('sha256')
async function copy(source,destination,filter,extra=[]){
  const paths=[...new Set([...execFileSync('git',['-C',source,'ls-files','-z'],{encoding:'utf8'}).split('\0').filter(Boolean),...extra])]
  for(const path of paths.filter(filter)){
    if(/(^|\/)\.(?:env[^/]*|git|npmrc)(\/|$)/.test(path))continue
    if(!(await lstat(join(source,path))).isFile())continue
    const to=join(root,'source',destination,path)
    await mkdir(dirname(to),{recursive:true});await copyFile(join(source,path),to)
    digest.update(destination+'/'+path+'\0').update(await readFile(to));files++
  }
}
