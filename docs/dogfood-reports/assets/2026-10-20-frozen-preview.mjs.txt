import assert from 'node:assert/strict'
import {VM,VmCheckpoint,ReadonlyProvider,RealFSProvider,createHttpHooks} from '@earendil-works/gondolin'
import {readFile,writeFile,mkdir} from 'node:fs/promises'
import {execFileSync} from 'node:child_process'
import {existsSync} from 'node:fs'
import {performance} from 'node:perf_hooks'
const root=new URL('.',import.meta.url).pathname
process.env.GONDOLIN_DEBUG=''
const tag=process.argv[2]??'one'
assert(/^[a-z]+$/.test(tag))
await mkdir(root+tag+'-incoming',{recursive:true})
execFileSync('tar',['-cf',root+tag+'-incoming/snapshot.tar','-C',root+'source','.'])
let setup=true,vm,ingress,stopping=false
let serverOutput,serverResult,serverLog=''
const metrics={tag,startedAt:new Date().toISOString(),phases:{}}
const start=performance.now()
const {httpHooks}=createHttpHooks({allowedHosts:['registry.npmjs.org','dl-cdn.alpinelinux.org'],blockInternalRanges:true,isRequestAllowed:r=>setup&&['GET','HEAD'].includes(r.method)&&['registry.npmjs.org','dl-cdn.alpinelinux.org'].includes(new URL(r.url).hostname)})
const options=()=>({debug:false,memory:'3G',cpus:2,rootfs:{mode:'cow',size:'8G'},tmpfs:{'/tmp':{size:'256m'}},vfs:{mounts:{'/incoming':new ReadonlyProvider(new RealFSProvider(root+tag+'-incoming'))}},httpHooks,startTimeoutMs:60000,sandbox:{qemuPath:'/opt/homebrew/bin/qemu-system-aarch64'},sessionLabel:'CIVIC frozen Laundry spike'})
let stop
const stopped=new Promise(resolve=>{stop=resolve})
process.once('SIGTERM',()=>{stopping=true;stop()})
const run=async command=>{const r=await vm.exec(command);assert.equal(r.exitCode,0,r.stderr+'\n'+r.stdout);return r.stdout}
try{
  const prepared=existsSync(root+'ready.qcow2')
  vm=prepared?await VmCheckpoint.load(root+'ready.qcow2').resume(options()):await VM.create(options())
  await run('true');metrics.phases.bootReadyMs=performance.now()-start
  console.log('BOOT_READY',metrics.phases.bootReadyMs)
  if(!prepared){
    console.log('DEPENDENCIES_BEGIN')
    const begin=performance.now()
    const install=vm.exec('mkdir -p /work/project; tar -xf /incoming/snapshot.tar -C /work/project; npm install --global pnpm@9.15.9 --ignore-scripts --no-audit --no-fund && cd /work/project && pnpm install --ignore-scripts --no-frozen-lockfile && apk add --no-cache chromium',{stdout:'pipe',stderr:'pipe'})
    for await(const{stream,text}of install.output())process.stdout.write(stream+': '+text)
    assert.equal((await install).exitCode,0)
    metrics.phases.provisionMs=performance.now()-begin
    const lock=await run('cat /work/project/pnpm-lock.yaml')
    await writeFile(root+'linux-pnpm-lock.yaml',lock)
    const checkpoint=await vm.checkpoint(root+'ready.qcow2')
    vm=await checkpoint.resume(options())
    console.log('DEPENDENCIES_READY',metrics.phases.provisionMs)
  }
  setup=false
  const copyStart=performance.now()
  await run('tar -xf /incoming/snapshot.tar -C /work/project; mkdir -p /work/project/adapter/node_modules/.pnpm /work/store /work/project/node_modules/.vite; mount --bind /work/project/node_modules/.pnpm /work/project/adapter/node_modules/.pnpm; chown -R 65534:65534 /work/store /work/project/node_modules/.vite')
  await run('mkdir -p /work/project/node_modules/@playwright; ln -sfn /work/project/civic/packages/canvas/node_modules/@playwright/test /work/project/node_modules/@playwright/test')
  for(const path of ['civic','laundry','adapter','journey-engine'])await run(`mount --bind /work/project/${path} /work/project/${path}; mount -o remount,bind,ro /work/project/${path}`)
  const readonly=await vm.exec('su -s /bin/sh nobody -c "echo forbidden >> /work/project/laundry/package.json"')
  assert.notEqual(readonly.exitCode,0);metrics.readonly=readonly.stderr
  const rootWrite=await vm.exec('echo forbidden >> /work/project/laundry/package.json')
  assert.notEqual(rootWrite.exitCode,0);metrics.readonlyRoot=rootWrite.stderr
  const egress=await vm.exec('wget -S -O- -T 5 https://example.com')
  assert.notEqual(egress.exitCode,0);metrics.egress={exitCode:egress.exitCode,stderr:egress.stderr}
  metrics.phases.copyAndFreezeMs=performance.now()-copyStart
  const server=vm.exec("su -s /bin/sh nobody -c 'cd /work/project && node guest.mjs'",{stdout:'pipe',stderr:'pipe'})
  serverResult=Promise.resolve(server).then(result=>({result}),error=>({error:error.message}))
  serverOutput=(async()=>{for await(const{text}of server.output()){serverLog=(serverLog+text).slice(-8000);await writeFile(root+tag+'-server.log',serverLog)}})().catch(error=>{if(!stopping)console.log('SERVER_OUTPUT',error.message)})
  ingress=await vm.enableIngress({listenHost:'127.0.0.1',listenPort:0,upstreamHeaderTimeoutMs:60000,upstreamResponseTimeoutMs:60000});vm.setIngressRoutes([{prefix:'/',port:8470,stripPrefix:false}])
  await writeFile(root+'running.json',JSON.stringify({origin:ingress.url,controller:process.pid,runner:vm.getHostPid()}))
  const deadline=performance.now()+30000
  let attempts=0
  while(true){
    try{
      const session=await fetch(ingress.url+'/review',{signal:AbortSignal.timeout(1000)})
      const cookie=session.headers.getSetCookie().map(value=>value.split(';')[0]).join('; ')
      const r=await fetch(ingress.url+'/@fs/work/project/laundry/apps/web/src/civic-snapshot.css?direct',{headers:{cookie},signal:AbortSignal.timeout(1000)})
      const text=await r.text()
      if(r.ok){metrics.marker=text;break}
      if(attempts++%10===0)console.log('HEALTH',session.status,r.status,text.slice(0,200))
    }catch(error){if(attempts++<3)console.log('HEALTH_ERROR',error.message)}
    if(performance.now()>deadline)throw Error('Server startup timed out: '+serverLog)
    await new Promise(resolve=>setTimeout(resolve,100))
  }
  metrics.phases.serverReadyMs=performance.now()-start
  await writeFile(root+tag+'.json',JSON.stringify({origin:ingress.url,controller:process.pid,runner:vm.getHostPid(),metrics},null,2))
  console.log('READY',ingress.url,metrics.phases)
  await stopped
}finally{
  const before=performance.now()
  if(vm)await vm.close()
  if(serverOutput)await serverOutput
  if(serverResult)await serverResult
  if(ingress)await ingress.close()
  metrics.shutdownMs=performance.now()-before
  await writeFile(root+tag+'-metrics.json',JSON.stringify(metrics,null,2))
  console.log('CLOSED',metrics.shutdownMs,stopping)
}
