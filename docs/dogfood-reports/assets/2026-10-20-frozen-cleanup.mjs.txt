import assert from 'node:assert/strict'
import {readFile,writeFile,realpath,stat,rm} from 'node:fs/promises'
import {execFileSync} from 'node:child_process'
import {dirname} from 'node:path'
const target='/private/tmp/civic-gondolin-frozen-seAkVh'
const cwd=process.cwd(),root=await realpath(target)
assert.equal(root,target);assert.equal(dirname(root),'/private/tmp');assert.notEqual(root,cwd)
assert((await readFile(root+'/prepare.mjs','utf8')).includes('/Users/taymoor/Projects/CIVIC'))
const run=JSON.parse(await readFile(root+'/running.json','utf8'))
const command=pid=>{try{return execFileSync('ps',['-p',String(pid),'-o','command='],{encoding:'utf8'}).trim()}catch{return ''}}
assert(command(run.controller).includes(root+'/preview.mjs two'),'Only close the owned spike controller')
const runnerCommand=command(run.runner),files=[...runnerCommand.matchAll(/(?:file=|path=|unix:)(\/tmp\/gondolin-[^,\s]+)/g)].map(m=>m[1])
const sizeKiB=Number(execFileSync('du',['-sk',root],{encoding:'utf8'}).split(/\s+/)[0])
const start=performance.now();process.kill(run.controller,'SIGTERM')
while(command(run.controller)&&performance.now()-start<40000)await new Promise(r=>setTimeout(r,200))
assert.equal(command(run.controller),'');assert.equal(command(run.runner),'')
const metrics=JSON.parse(await readFile(root+'/two-metrics.json','utf8'))
const remains=[];for(const file of files)if(await stat(file).then(()=>true,()=>false))remains.push(file)
assert.deepEqual(remains,[],'Gondolin removes its COW disk and sockets')
assert.equal(await fetch(run.origin+'/review',{signal:AbortSignal.timeout(1000)}).then(()=>true,()=>false),false)
const output='/Users/taymoor/Projects/CIVIC/docs/dogfood-reports/assets/'
await writeFile(output+'2026-10-20-frozen-refresh-metrics.json',JSON.stringify(metrics,null,2))
await rm(root,{recursive:true})
assert.equal(await stat(root).then(()=>true,()=>false),false)
await writeFile(output+'2026-10-20-frozen-cleanup.json',JSON.stringify({cwd,root,controller:run.controller,runner:run.runner,reclaimedMiB:sizeKiB/1024,shutdownMs:metrics.shutdownMs,cleanupWallMs:performance.now()-start,portClosed:true,remainingOwnedRuntimeFiles:remains,sourceSnapshotRemoved:true,retained:'QEMU installation and pre-existing Gondolin asset cache'},null,2))
console.log('Disposable guest, controller, ingress, disk, sockets and snapshot removed; evidence retained.')
